PRICE CALCULATION (app/api/bookings/route.ts):
-------
if (data.specialServiceId && data.specialServiceName) {
  // Validate & use special service price
  const service = instructor.specialServices.find(s => s.id === data.specialServiceId)
  if (!service || !service.price) return error()
  lessonPrice = service.price
} else {
  // Standard hourly lesson
  lessonPrice = instructor.hourlyRate * durationHours
}
